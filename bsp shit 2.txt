		std::vector<std::vector<LineMap>> vectorPositions;

		int previousPathId = 0;

		std::vector<LineMap> lines;
		std::vector<LineMap> foundLines;

		map->drawLine(vectorsMapArray[0].startPosX, vectorsMapArray[0].startPosY, vectorsMapArray[0].endPosX, vectorsMapArray[0].endPosY, 255, 255, 0);
		SDL_RenderPresent(renderer);
		std::this_thread::sleep_for(std::chrono::seconds(1));

		lines.push_back(vectorsMapArray[0]);
		vectorsMapArray.erase(vectorsMapArray.begin() + 0);

		sectorPaths[pathDefaultName + std::to_string(previousPathId)] = lines;

		int j = 0;
		while (vectorsMapArray.size() >= 0)
		{
			if (j < sectorPaths.size())
			{
				std::string currentPathName = pathDefaultName + std::to_string(j);
				std::vector<LineMap>& tempLineMapComponent = sectorPaths[currentPathName];

				std::cout << "checking sector #" << j << std::endl;

				std::cout << "path arr size: " << sectorPaths.size() << std::endl;

				LineMap previousLineMapComponent = *new LineMap(); //uniquePtr

				for (int i = 0; i < vectorsMapArray.size(); i++)
				{
					if (vectorsMapArray.size() > 0)
					{
						std::cout << "map array size: " << vectorsMapArray.size() << std::endl;

						if (((vectorsMapArray[i].endPosX == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX) &&
							(vectorsMapArray[i].endPosY == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY)))
						{
							double tempEndPosX = vectorsMapArray[i].startPosX;
							double tempEndPosY = vectorsMapArray[i].startPosY;

							vectorsMapArray[i].startPosX = vectorsMapArray[i].endPosX;
							vectorsMapArray[i].startPosY = vectorsMapArray[i].endPosY;
							vectorsMapArray[i].endPosX = tempEndPosX;
							vectorsMapArray[i].endPosY = tempEndPosY;
							vectorsMapArray[i].dirX = -(vectorsMapArray[i].endPosX);
							vectorsMapArray[i].dirX = -(vectorsMapArray[i].endPosY);

							foundLines.push_back(vectorsMapArray[i]);

							std::cout << "adding vector start pos: " << vectorsMapArray[i].startPosX << ", " << vectorsMapArray[i].startPosY << std::endl;

							vectorsMapArray.erase(vectorsMapArray.begin() + i);
							i = -1;
						}
						else if (((vectorsMapArray[i].startPosX == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX)
							&& (vectorsMapArray[i].startPosY == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY)))
						{
							foundLines.push_back(vectorsMapArray[i]);

							std::cout << "adding vector start pos: " << vectorsMapArray[i].startPosX << ", " << vectorsMapArray[i].startPosY << std::endl;

							vectorsMapArray.erase(vectorsMapArray.begin() + i);
							i = -1;
						}
					}
				}

				if (foundLines.size() > 1)
				{
					std::cout << "creating new paths ";

					//call metthod to sort the array
					quickSort(foundLines, 0, foundLines.size() - 1, tempLineMapComponent[tempLineMapComponent.size() - 1]);

					for (int i = 0; i < foundLines.size(); i++)
					{
						previousPathId += 1;

						std::string pathName = pathDefaultName + std::to_string(previousPathId);
						std::cout << pathName << std::endl;

						sectorPaths[pathName].push_back(foundLines[i]);

						map->drawLine(foundLines[i].startPosX, foundLines[i].startPosY, foundLines[i].endPosX, foundLines[i].endPosY, 255, 255, 0);
						SDL_RenderPresent(renderer);
						std::this_thread::sleep_for(std::chrono::seconds(1));

						//делаем проверку, являяется ли конечная точка добавленного вектора начальной точкой для более одного вектора
						//previousPathId = checkAmountOfPaths(foundLines[i], previousPathId);
					}
					foundLines.clear();
					j++;
				}
				else
				{
					int flag = checkPath(sectorPaths, tempLineMapComponent[tempLineMapComponent.size() - 1]);

					std::cout << "true or not: " << flag << std::endl;

					if (flag == -1)
					{
						if (foundLines.size() == 1)
						{
							std::cout << "adding new line" << std::endl;
							std::cout << currentPathName << std::endl;
							std::string pathName = currentPathName;

							sectorPaths[pathName].push_back(foundLines[0]);

							map->drawLine(foundLines[0].startPosX, foundLines[0].startPosY, foundLines[0].endPosX, foundLines[0].endPosY, 255, 255, 0);
							SDL_RenderPresent(renderer);
							std::this_thread::sleep_for(std::chrono::seconds(1));
						}
						else
						{
							j++;
						}
					}
					else
					{
						std::cout << "creating new sector" << std::endl;

						tempLineMapComponent[tempLineMapComponent.size() - 1].endOfSector = true;

						std::vector<LineMap>& foundPath = sectorPaths[pathDefaultName + std::to_string(flag)];

						if ((tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX == foundPath[0].startPosX) &&
							(tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY == foundPath[0].startPosY))
						{
							std::vector<LineMap> fullSection;

							if ((tempLineMapComponent[tempLineMapComponent.size() - 1].startPosX != foundPath[0].endPosX) &&
								(tempLineMapComponent[tempLineMapComponent.size() - 1].startPosY != foundPath[0].endPosY))
							{
								LineMap tempLine = *new LineMap(); //uniquePtr
								std::vector<LineMap>& tempPath = tempLineMapComponent;

								std::cout << "1 line with XEnd,Yend,XStart,YStart has been added: " << tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX << ", " << tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY << ", " <<
									tempLineMapComponent[tempLineMapComponent.size() - 1].startPosX << ", " << tempLineMapComponent[tempLineMapComponent.size() - 1].startPosY << std::endl;

								fullSection.insert(fullSection.end(), tempPath.begin(), tempPath.end());

								tempLine.endPosX = tempPath[0].startPosX;
								tempLine.endPosY = tempPath[0].startPosY;

								int numerLine = 2;

								while ((tempLine.endPosX != foundPath[foundPath.size() - 1].endPosX) &&
									(tempLine.endPosY != foundPath[foundPath.size() - 1].endPosY))
								{
									for (int n = 0; n < sectorPaths.size(); n++)
									{
										tempPath = sectorPaths[pathDefaultName + std::to_string(n)];
										if ((tempPath[tempPath.size() - 1].endPosX == tempLine.endPosX) &&
											(tempPath[tempPath.size() - 1].endPosY == tempLine.endPosY))
										{
											fullSection.insert(fullSection.end(), tempPath.begin(), tempPath.end());

											std::cout << numerLine << " line with XEnd,Yend,XStart,YStart has been added: " << tempPath[tempLineMapComponent.size() - 1].endPosX << ", " << tempPath[tempLineMapComponent.size() - 1].endPosY << ", " <<
												tempPath[0].startPosX << ", " << tempPath[0].startPosY << std::endl;

											numerLine++;

											tempLine.endPosX = tempPath[0].startPosX;
											tempLine.endPosY = tempPath[0].startPosY;
											break;
										}
									}
								}
							}
							fullSection.insert(fullSection.end(), foundPath.begin(), foundPath.end());
							vectorPositions.push_back(fullSection);
						}
						else
						{
							std::vector<LineMap> fullSection;

							LineMap lineFirst = *new LineMap(); //uniquePtr
							LineMap lineSecond = *new LineMap(); //uniquePtr

							std::vector<std::vector<LineMap>> pathFirst;
							std::vector<std::vector<LineMap>> pathSecond;
							
							std::vector<LineMap> tempPath;
							
							tempPath = sectorPaths[pathDefaultName + std::to_string(flag)];

							lineFirst.startPosX = tempLineMapComponent[0].startPosX;
							lineFirst.startPosY = tempLineMapComponent[0].startPosY;

							lineSecond.startPosX = tempPath[0].startPosX;
							lineSecond.startPosY = tempPath[0].startPosY;

							std::vector<PathEndPosition> pathEndFirstPositions;
							std::vector<PathEndPosition> pathEndSecondPositions;

							PathEndPosition pathFirstPosition = *new PathEndPosition(lineFirst.startPosX, lineFirst.startPosY); //uniquePtr
							PathEndPosition pathSecondPosition = *new PathEndPosition(lineSecond.startPosX, lineSecond.startPosY); //uniquePtr

							pathEndFirstPositions.push_back(pathFirstPosition);
							pathEndSecondPositions.push_back(pathSecondPosition);

							pathFirst.push_back(tempLineMapComponent);
							pathSecond.push_back(tempPath);

							bool sectorIsClosed = false;

							while (!sectorIsClosed)
							{
								if (!sectorIsClosed)
								{
									for (int n = 0; n < sectorPaths.size(); n++)
									{
										if (tempPath.size() != 0)
										{
											tempPath.clear();
										}

										tempPath = sectorPaths[pathDefaultName + std::to_string(n)];
										if ((lineFirst.startPosX == tempPath[pathFirst.size() - 1].endPosX) &&
											(lineFirst.startPosY == tempPath[pathFirst.size() - 1].endPosY))
										{
											lineFirst.startPosX = tempPath[0].startPosX;
											lineFirst.startPosY = tempPath[0].startPosY;

											PathEndPosition path = *new PathEndPosition(lineFirst.startPosX, lineFirst.startPosY); //uniquePtr

											int indexToClear = checkPathPos(pathEndSecondPositions, path);

											if (indexToClear == -1)
											{
												pathEndFirstPositions.push_back(path);
												pathFirst.push_back(tempPath);
											}
											else
											{
												pathFirst.push_back(tempPath);
												pathSecond.erase(pathSecond.begin() + indexToClear, pathSecond.end());

												sectorIsClosed = true;
											}

											break;
										}
									}
								}

								if (!sectorIsClosed)
								{
									for (int n = 0; n < sectorPaths.size(); n++)
									{
										if (tempPath.size() != 0)
										{
											tempPath.clear();
										}

										tempPath = sectorPaths[pathDefaultName + std::to_string(n)];
										if ((lineSecond.startPosX == tempPath[pathSecond.size() - 1].endPosX) &&
											(lineSecond.startPosY == tempPath[pathSecond.size() - 1].endPosY))
										{

											lineSecond.startPosX = tempPath[0].startPosX;
											lineSecond.startPosY = tempPath[0].startPosY;

											PathEndPosition path = *new PathEndPosition(lineFirst.startPosX, lineFirst.startPosY);

											int indexToClear = checkPathPos(pathEndFirstPositions, path);

											if (indexToClear == -1)
											{
												pathEndSecondPositions.push_back(path);
												pathSecond.push_back(tempPath);
											}
											else
											{
												pathSecond.push_back(tempPath);
												pathFirst.erase(pathFirst.begin() + indexToClear, pathFirst.end());

												sectorIsClosed = true;
											}

											break;
										}
									}
								}
							}

							for (int i = 0; i < pathFirst.size(); i++)
							{
								fullSection.insert(fullSection.end(), pathFirst[i].begin(), pathFirst[i].end());
							}

							for (int i = 0; i < pathSecond.size(); i++)
							{
								fullSection.insert(fullSection.end(), pathSecond[i].begin(), pathSecond[i].end());
							}

							vectorPositions.push_back(fullSection);
						}
						j++;
						if (foundLines.size() > 0)
						{
							vectorsMapArray.push_back(foundLines[0]);
						}
					}
					std::cout << "sector has been created" << std::endl;
					foundLines.clear();
				}
			}
			else
			{
				break;
			}
		}
		std::cout << "vectorPositions size: " << vectorPositions.size() << std::endl;
		for (int i = 0; i < vectorPositions.size(); i++)
		{
			std::vector<LineMap>& pathFirst = vectorPositions[i];

			std::cout << "vectorPositions inner size: " << pathFirst.size() << std::endl;

			for (int j = 0; j < pathFirst.size(); j++)
			{
				map->drawLine(pathFirst[j].startPosX, pathFirst[j].startPosY, pathFirst[j].endPosX, pathFirst[j].endPosY, 0, 255, 0);
				SDL_RenderPresent(renderer);
			}

			std::this_thread::sleep_for(std::chrono::seconds(5));
			SDL_SetRenderDrawColor(renderer, 0, 0, 0, 0);
			SDL_RenderClear(renderer);
		}
