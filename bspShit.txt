	std::vector<std::vector<lineMap>> vectorPositions;

	int generalDotsCounter = 0;
	int previousPathId = 0;

	std::map<std::string, std::vector<lineMap>> sectorPaths;
	std::string pathDefaultName = "path";

	std::vector<lineMap> lines;

	vectorsMapArray[0].wasRead = true;
	lines.push_back(vectorsMapArray[0]);

	sectorPaths[pathDefaultName + std::to_string(previousPathId)] = lines;

	for (int j = 0; j < sectorPaths.size(); j++)
	{
		std::string currentPathName = pathDefaultName + std::to_string(j);
		std::vector<lineMap>& tempLineMapComponent = sectorPaths[currentPathName];

		bool isWritable = false;
		generalDotsCounter = 0;

		for (int i = 0; i < vectorsMapArray.size(); i++)
		{
			if (((vectorsMapArray[i].endPosX == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX) &&
				(vectorsMapArray[i].endPosY == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY)))
			{
				double tempEndPosX = vectorsMapArray[i].startPosX;
				double tempEndPosY = vectorsMapArray[i].startPosY;

				vectorsMapArray[i].startPosX = vectorsMapArray[i].endPosX;
				vectorsMapArray[i].startPosY = vectorsMapArray[i].endPosY;
				vectorsMapArray[i].endPosX = tempEndPosX;
				vectorsMapArray[i].endPosY = tempEndPosY;
				vectorsMapArray[i].dirX = -(vectorsMapArray[i].endPosX);
				vectorsMapArray[i].dirX = -(vectorsMapArray[i].endPosY);

				isWritable = true;
			}
			else if (((vectorsMapArray[i].startPosX == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX)
				&& (vectorsMapArray[i].startPosY == tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY)))
			{
				isWritable = true;
			}
			//переделать логику отработки точек и итератора
			//проверять не каждый вектор, а начало path и конец path и так каждый path - минимизируем лишние итерации
			if (isWritable)
			{
				if (vectorsMapArray[i].wasRead == false)
				{
					generalDotsCounter++;
					if (generalDotsCounter >= 2)
					{
						previousPathId += 1;

						lines.clear();

						std::string pathName = pathDefaultName + std::to_string(previousPathId);

						vectorsMapArray[i].wasRead = true;
						vectorsMapArray[i].pathName = pathName;

						lines.push_back(vectorsMapArray[i]);
						sectorPaths[pathName] = lines;

						map->drawLine(vectorsMapArray[i].startPosX, vectorsMapArray[i].startPosY, vectorsMapArray[i].endPosX, vectorsMapArray[i].endPosY, 255, 255, 0);
						SDL_RenderPresent(renderer);
						std::this_thread::sleep_for(std::chrono::seconds(1));
					}
					else if (generalDotsCounter == 1)
					{
						std::string pathName = pathDefaultName + std::to_string(previousPathId);

						vectorsMapArray[i].wasRead = true;
						vectorsMapArray[i].pathName = pathName;

						std::cout << "adding new line to " << pathName << std::endl;
						std::cout << "size of the sector " << currentPathName << " : " << tempLineMapComponent.size() << std::endl;

						sectorPaths[pathName].push_back(vectorsMapArray[i]);

						map->drawLine(vectorsMapArray[i].startPosX, vectorsMapArray[i].startPosY, vectorsMapArray[i].endPosX, vectorsMapArray[i].endPosY, 255, 255, 0);
						SDL_RenderPresent(renderer);
						std::this_thread::sleep_for(std::chrono::seconds(1));

						generalDotsCounter = 0;
					}
				}
				else
				{
					std::cout << "size of the sector " << currentPathName << " : " << tempLineMapComponent.size() << std::endl;

					if (tempLineMapComponent.size() > 1)
					{
						if ((tempLineMapComponent[tempLineMapComponent.size() - 1].endPosX == vectorsMapArray[i].startPosX) &&
							(tempLineMapComponent[tempLineMapComponent.size() - 1].endPosY == vectorsMapArray[i].startPosY))
						{
							std::vector<lineMap> fullSection;

							lineMap tempLine = *new lineMap();
							std::vector<lineMap>& tempPath = tempLineMapComponent;

							fullSection.insert(fullSection.end(), tempPath.begin(), tempPath.end());

							tempLine.endPosX = tempPath[0].startPosX;
							tempLine.endPosY = tempPath[0].startPosY;

							while ((tempLine.endPosX != vectorsMapArray[i].endPosX) &&
								(tempLine.endPosY != vectorsMapArray[i].endPosY))
							{

								for (int n = 0; n < sectorPaths.size(); n++)
								{
									tempPath = sectorPaths[pathDefaultName + std::to_string(n)];
									if ((tempPath[tempPath.size() - 1].endPosX == tempLine.endPosX) &&
										(tempPath[tempPath.size() - 1].endPosY == tempLine.endPosY))
									{
										fullSection.insert(fullSection.end(), tempPath.begin(), tempPath.end());

										tempLine.endPosX = tempPath[0].startPosX;
										tempLine.endPosY = tempPath[0].startPosY;
										break;
									}
								}
							}
							vectorPositions.push_back(fullSection);
						}
						else
						{
							std::vector<lineMap> fullSection;

							lineMap lineFirst = *new lineMap();
							lineMap lineSecond = *new lineMap();

							std::vector<lineMap>& pathFirst = tempLineMapComponent;
							std::vector<lineMap>& pathSecond = sectorPaths[vectorsMapArray[i].pathName];

							lineFirst.startPosX = pathFirst[0].startPosX;
							lineFirst.startPosY = pathFirst[0].startPosY;

							lineSecond.startPosX = pathSecond[0].startPosX;
							lineSecond.startPosY = pathSecond[0].startPosY;

							while ((lineFirst.startPosX != lineSecond.startPosX) &&
								lineFirst.startPosY != lineSecond.startPosY)
							{
								for (int n = 0; n < sectorPaths.size(); n++)
								{
									pathFirst = sectorPaths[pathDefaultName + std::to_string(n)];
									if ((lineFirst.startPosX == pathFirst[pathFirst.size() - 1].endPosX) &&
										(lineFirst.startPosY == pathFirst[pathFirst.size() - 1].endPosY))
									{
										fullSection.insert(fullSection.end(), pathFirst.begin(), pathFirst.end());

										lineFirst.startPosX = pathFirst[0].startPosX;
										lineFirst.startPosY = pathFirst[0].startPosY;
										break;
									}
								}

								for (int n = 0; n < sectorPaths.size(); n++)
								{
									pathSecond = sectorPaths[pathDefaultName + std::to_string(n)];
									if ((lineSecond.startPosX == pathSecond[pathSecond.size() - 1].endPosX) &&
										(lineSecond.startPosY == pathSecond[pathSecond.size() - 1].endPosY))
									{
										fullSection.insert(fullSection.end(), pathSecond.begin(), pathSecond.end());

										lineSecond.startPosX = pathSecond[0].startPosX;
										lineSecond.startPosY = pathSecond[0].startPosY;
										break;
									}
								}
							}
							vectorPositions.push_back(fullSection);
						}
					}
				}
				isWritable = false;
			}
		}
	}
	std::cout << vectorPositions.size() << std::endl;
	for (int i = 0; i < vectorPositions.size(); i++)
	{
		std::vector<lineMap>& tempArray = vectorPositions[i];

		for (int j = 0; j < tempArray.size(); j++)
		{
			map->drawLine(tempArray[j].startPosX, tempArray[j].startPosY, tempArray[j].endPosX, tempArray[j].endPosY, 0, 255, 0);
			SDL_RenderPresent(renderer);
		}
	}
	std::this_thread::sleep_for(std::chrono::seconds(5));
}
